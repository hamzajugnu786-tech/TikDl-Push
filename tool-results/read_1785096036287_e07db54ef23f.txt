   107→const TikTokDownloader = () => {
   108→  const [url, setUrl] = useState('');
   109→  const [isLoading, setIsLoading] = useState(false);
   110→  const [videoInfo, setVideoInfo] = useState<VideoInfo | null>(null);
   111→  const [error, setError] = useState('');
   112→  const [history, setHistory] = useState<VideoInfo[]>([]);
   113→  const [activeTab, setActiveTab] = useState<'video' | 'audio' | 'cover'>('video');
   114→  const [showAdPopup, setShowAdPopup] = useState(false);
   115→  const [countdown, setCountdown] = useState(5);
   116→  const [pendingUrl, setPendingUrl] = useState('');
   117→  const [openFAQ, setOpenFAQ] = useState<number | null>(null);
   118→  const [autoProceedDone, setAutoProceedDone] = useState(false);
   119→  const [interstitialConfig, setInterstitialConfig] = useState({
   120→    enabled: true,
   121→    countdownDuration: 5,
   122→    autoDownload: true,
   123→    popupTitle: 'Support free downloads',
   124→    popupDescription: 'Your download will start automatically...',
   125→  });
   126→  const adTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   127→  const countdownRef = useRef<number>(5);
   128→  const autoProceedRef = useRef(false);
   129→
   130→  // Fetch interstitial config on mount
   131→  useEffect(() => {
   132→    const fetchConfig = async () => {
   133→      try {
   134→        const res = await fetch('/api/config/ads');
   135→        const data = await res.json();
   136→        if (data.success && data.interstitial) {
   137→          setInterstitialConfig(data.interstitial);
   138→        }
   139→      } catch {
   140→        // Use default config on fetch failure
   141→      }
   142→    };
   143→    fetchConfig();
   144→  }, []);
   145→
   146→  // Load history from localStorage
   147→  useEffect(() => {
   148→    try {
   149→      const saved = localStorage.getItem('tiktokHistory');
   150→      if (saved) {
   151→        const parsed = JSON.parse(saved);
   152→        if (Array.isArray(parsed)) setHistory(parsed);
   153→      }
   154→    } catch (e) {
   155→      console.warn('Failed to load history:', e);
   156→    }
   157→  }, []);
   158→
   159→  // Save history to localStorage
   160→  useEffect(() => {
   161→    try {
   162→      localStorage.setItem('tiktokHistory', JSON.stringify(history));
   163→    } catch (e) {
   164→      console.warn('Failed to save history:', e);
   165→    }
   166→  }, [history]);
   167→

   169→    const regex = /^https?:\/\/(?:www\.|vm\.|vt\.|m\.)?tiktok\.com\/.+/i;
   170→    return regex.test(inputUrl.trim());

   175→    setCountdown(duration);
   176→    setAutoProceedDone(false);
   177→    autoProceedRef.current = false;
   178→    setShowAdPopup(true);
   179→    setPendingUrl(videoUrl);
   180→    countdownRef.current = duration;
   181→
   182→    if (adTimerRef.current) clearInterval(adTimerRef.current);
   183→
   184→    adTimerRef.current = setInterval(() => {
   185→      countdownRef.current -= 1;
   186→      setCountdown(countdownRef.current);
   187→
   188→      if (countdownRef.current <= 0) {
   189→        if (adTimerRef.current) clearInterval(adTimerRef.current);
   190→        adTimerRef.current = null;
   191→      }
   192→    }, 1000);
   193→  }, [interstitialConfig.countdownDuration]);
   194→
   195→  const proceedAfterAd = useCallback(async () => {
   196→    setShowAdPopup(false);
   197→
   198→    const videoUrl = pendingUrl;
   199→    if (!videoUrl) return;
   200→
   201→    setIsLoading(true);
   202→    setError('');
   203→    setVideoInfo(null);
   204→
   205→    try {
   206→      const response = await fetch('/api/download', {
   207→        method: 'POST',
   208→        headers: { 'Content-Type': 'application/json' },
   209→        body: JSON.stringify({ url: videoUrl }),
   210→      });
   211→
   212→      const result = await response.json();
   213→
   214→      if (!result.success) {
   215→        throw new Error(result.error || 'Failed to fetch video info');
   216→      }
   217→
   218→      const data: VideoInfo = result.data;
   219→      setVideoInfo(data);
   220→      setHistory(prev => [data, ...prev.slice(0, 4)]);
   221→
   222→      toast.success('Video ready!', {
   223→        description: `Fetched via ${result.provider} in ${result.duration}ms`,
   224→      });
   225→    } catch (err: unknown) {
   226→      const errorMessage = err instanceof Error ? err.message : 'Unable to process this video. Try again.';
   227→      console.error('Download error:', err);
   228→      setError(errorMessage);
   229→      toast.error('Fetch failed', { description: errorMessage });

   231→      setIsLoading(false);
   232→      setPendingUrl('');
   233→    }
   234→  }, [pendingUrl]);
   235→

   237→  useEffect(() => {

   240→      setAutoProceedDone(true);
   241→      const timeout = setTimeout(() => {
   242→        proceedAfterAd();
   243→      }, 800);
   244→      return () => clearTimeout(timeout);
   245→    }
   246→  }, [countdown, showAdPopup, interstitialConfig.autoDownload, proceedAfterAd]);
   247→
   248→  const handleSubmit = useCallback((e: React.FormEvent) => {

   258→      return;
   259→    }
   260→    setError('');
   261→    startAdTimer(trimmed);
   262→  }, [url, startAdTimer]);
   263→

   267→      return;

   273→    a.download = filename;
   274→    a.target = '_blank';
   275→    a.rel = 'noopener noreferrer';
   276→    document.body.appendChild(a);
   277→    a.click();

   279→
   280→    toast.success(`Downloading ${filename}`, {
   281→      description: 'Your file download has started.',
   282→    });
   283→  }, []);

   292→    }
   293→  }, []);

   313→    };
   314→  }, []);

   317→    <div className="min-h-screen bg-[#000000] text-white flex flex-col">
   318→      <Toaster position="top-center" richColors closeButton />

   322→        <div className="max-w-7xl mx-auto px-4 sm:px-6 py-4 flex justify-between items-center">
   323→          <div className="flex items-center gap-3">
   324→            <div className="w-8 h-8 bg-[#FE2C55] rounded-xl flex items-center justify-center text-xl font-bold">♪</div>
   325→            <span className="font-bold text-2xl sm:text-3xl tracking-tighter">TikDL</span>
   326→          </div>

   328→            <a href="#features" className="hover:text-[#FE2C55] transition-colors">Features</a>
   329→            <a href="#faq" className="hover:text-[#FE2C55] transition-colors">FAQ</a>
   330→            <a href="#history" className="hover:text-[#FE2C55] transition-colors">History</a>

   418→          {/* Supported formats hint */}

   423→            className="mt-8 flex flex-wrap justify-center gap-6 text-sm text-[#888]"
   424→          >

   426→            <span className="flex items-center gap-1.5"><Music size={15} className="text-[#25F4EE]" /> MP3 Audio</span>
