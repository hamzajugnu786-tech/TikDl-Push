   107→const TikTokDownloader = () => {
   108→  const [url, setUrl] = useState('');
   109→  const [isLoading, setIsLoading] = useState(false);
   110→  const [videoInfo, setVideoInfo] = useState<VideoInfo | null>(null);
   111→  const [error, setError] = useState('');
   112→  const [history, setHistory] = useState<VideoInfo[]>([]);
   113→  const [activeTab, setActiveTab] = useState<'video' | 'audio' | 'cover'>('video');
   114→  const [showAdPopup, setShowAdPopup] = useState(false);
   115→  const [countdown, setCountdown] = useState(5);
   116→  const [isAdComplete, setIsAdComplete] = useState(false);
   117→  const [pendingUrl, setPendingUrl] = useState('');
   118→  const [openFAQ, setOpenFAQ] = useState<number | null>(null);
   119→  const adTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   120→  const countdownRef = useRef<number>(5);
   121→  const isAdCompleteRef = useRef(false);
   122→
   123→  // Load history from localStorage
   124→  useEffect(() => {
   125→    try {
   126→      const saved = localStorage.getItem('tiktokHistory');
   127→      if (saved) {
   128→        const parsed = JSON.parse(saved);
   129→        if (Array.isArray(parsed)) setHistory(parsed);
   130→      }
   131→    } catch (e) {
   132→      console.warn('Failed to load history:', e);
   133→    }
   134→  }, []);
   135→
   136→  // Save history to localStorage
   137→  useEffect(() => {
   138→    try {
   139→      localStorage.setItem('tiktokHistory', JSON.stringify(history));
   140→    } catch (e) {
   141→      console.warn('Failed to save history:', e);
   142→    }
   143→  }, [history]);
   144→

   146→    const regex = /^https?:\/\/(?:www\.|vm\.|vt\.|m\.)?tiktok\.com\/.+/i;
   147→    return regex.test(inputUrl.trim());
   148→  };
   149→
   150→  const startAdTimer = useCallback((videoUrl: string) => {
   151→    // Reset state
   152→    setCountdown(5);
   153→    setIsAdComplete(false);
   154→    setShowAdPopup(true);
   155→    setPendingUrl(videoUrl);
   156→    countdownRef.current = 5;
   157→    isAdCompleteRef.current = false;
   158→
   159→    // Start countdown
   160→    if (adTimerRef.current) clearInterval(adTimerRef.current);
   161→
   162→    adTimerRef.current = setInterval(() => {
   163→      countdownRef.current -= 1;
   164→      setCountdown(countdownRef.current);
   165→
   166→      if (countdownRef.current <= 0) {
   167→        if (adTimerRef.current) clearInterval(adTimerRef.current);
   168→        adTimerRef.current = null;
   169→        setIsAdComplete(true);
   170→        isAdCompleteRef.current = true;
   171→      }
   172→    }, 1000);
   173→  }, []);
   174→
   175→  const skipAd = useCallback(() => {
   176→    if (adTimerRef.current) {
   177→      clearInterval(adTimerRef.current);
   178→      adTimerRef.current = null;
   179→    }
   180→    setIsAdComplete(true);
   181→    isAdCompleteRef.current = true;
   182→    setCountdown(0);
   183→  }, []);
   184→
   185→  const proceedAfterAd = useCallback(async () => {
   186→    setShowAdPopup(false);
   187→
   188→    const videoUrl = pendingUrl;
   189→    if (!videoUrl) return;
   190→
   191→    setIsLoading(true);
   192→    setError('');
   193→    setVideoInfo(null);
   194→
   195→    try {
   196→      const response = await fetch('/api/download', {
   197→        method: 'POST',
   198→        headers: { 'Content-Type': 'application/json' },
   199→        body: JSON.stringify({ url: videoUrl }),
   200→      });
   201→
   202→      const result = await response.json();
   203→
   204→      if (!result.success) {
   205→        throw new Error(result.error || 'Failed to fetch video info');
   206→      }
   207→
   208→      const data: VideoInfo = result.data;
   209→      setVideoInfo(data);
   210→      setHistory(prev => [data, ...prev.slice(0, 4)]);
   211→
   212→      toast.success('Video ready!', {
   213→        description: `Fetched via ${result.provider} in ${result.duration}ms`,
   214→      });
   215→    } catch (err: unknown) {
   216→      const errorMessage = err instanceof Error ? err.message : 'Unable to process this video. Try again.';
   217→      console.error('Download error:', err);
   218→      setError(errorMessage);
   219→      toast.error('Fetch failed', { description: errorMessage });
   220→    } finally {
   221→      setIsLoading(false);
   222→      setPendingUrl('');
   223→    }
   224→  }, [pendingUrl]);
   225→
   226→  const handleSubmit = useCallback((e: React.FormEvent) => {
   227→    e.preventDefault();

   229→    if (!trimmed) {
   230→      toast.error('Enter a TikTok URL');
   231→      return;
   232→    }
   233→    if (!isValidTikTokUrl(trimmed)) {
   234→      setError('Invalid TikTok URL format. Please use a valid TikTok link.');
   235→      toast.error('Please use a valid TikTok link');
   236→      return;
   237→    }
   238→    setError('');
   239→    startAdTimer(trimmed);
   240→  }, [url, startAdTimer]);
   241→
   242→  const handleDownload = useCallback((downloadUrl: string, filename: string) => {
   243→    if (!downloadUrl || downloadUrl.startsWith('#')) {
   244→      toast.error('Download URL not available for this video.');
   245→      return;
   246→    }
   247→
   248→    // Use a hidden anchor to trigger the actual download
   249→    const a = document.createElement('a');
   250→    a.href = downloadUrl;
   251→    a.download = filename;
   252→    a.target = '_blank';
   253→    a.rel = 'noopener noreferrer';
   254→    document.body.appendChild(a);
   255→    a.click();
   256→    document.body.removeChild(a);
   257→
   258→    toast.success(`Downloading ${filename}`, {
   259→      description: 'Your file download has started.',
   260→    });
   261→  }, []);
   262→
   263→  const getDownloadFilename = useCallback((type: 'video' | 'audio' | 'cover', info: VideoInfo): string => {
   264→    const sanitizeName = (name: string) => name.replace(/[^a-zA-Z0-9_-]/g, '_').slice(0, 50);
   265→    const baseName = sanitizeName(info.title || info.id);

   268→      case 'audio': return `tiktok_${baseName}_audio.mp3`;
   269→      case 'cover': return `tiktok_${baseName}_cover.jpg`;
   270→    }
   271→  }, []);

   278→    });
   279→  }, []);
   280→
   281→  const clearHistory = useCallback(() => {
   282→    setHistory([]);

   284→    toast.success('History cleared');

   290→      if (adTimerRef.current) clearInterval(adTimerRef.current);
   291→    };
   292→  }, []);
   293→
   294→  return (
   295→    <div className="min-h-screen bg-[#000000] text-white flex flex-col">
   296→      <Toaster position="top-center" richColors closeButton />

   298→      {/* ===== Navbar ===== */}
   299→      <nav className="sticky top-0 z-50 glass border-b border-white/10 bg-black/80">
   300→        <div className="max-w-7xl mx-auto px-4 sm:px-6 py-4 flex justify-between items-center">
   301→          <div className="flex items-center gap-3">
   302→            <div className="w-8 h-8 bg-[#FE2C55] rounded-xl flex items-center justify-center text-xl font-bold">♪</div>
   303→            <span className="font-bold text-2xl sm:text-3xl tracking-tighter">TikDL</span>
   304→          </div>
   305→          <div className="hidden md:flex gap-8 text-sm font-medium">
   306→            <a href="#features" className="hover:text-[#FE2C55] transition-colors">Features</a>
   307→            <a href="#faq" className="hover:text-[#FE2C55] transition-colors">FAQ</a>
   308→            <a href="#history" className="hover:text-[#FE2C55] transition-colors">History</a>
   309→          </div>
   310→        </div>
   311→      </nav>
   312→
   313→      {/* ===== Hero Section ===== */}
   314→      <section className="pt-16 sm:pt-20 pb-10 sm:pb-12 px-4 sm:px-6">
   315→        <div className="max-w-3xl mx-auto text-center">
   316→          <motion.h1
   317→            initial={{ opacity: 0, y: 20 }}
   318→            animate={{ opacity: 1, y: 0 }}
   319→            transition={{ duration: 0.6 }}
   320→            className="text-4xl sm:text-5xl md:text-6xl font-bold tracking-tighter mb-4 sm:mb-6"

   322→            TikTok Without Watermark

   324→          <motion.p
   325→            initial={{ opacity: 0, y: 10 }}
   326→            animate={{ opacity: 1, y: 0 }}
   327→            transition={{ duration: 0.6, delay: 0.1 }}
   328→            className="text-lg sm:text-xl text-gray-400 mb-8 sm:mb-10"
   329→          >

   332→
   333→          <motion.form
   334→            onSubmit={handleSubmit}

   337→            transition={{ duration: 0.5, delay: 0.2 }}
   338→            className="glass p-2 rounded-3xl flex flex-col sm:flex-row gap-3 max-w-xl mx-auto"

   341→              type="text"

   386→              >
   387→                {error}
   388→              </motion.div>
   389→            )}

   391→
   392→          {/* Supported formats hint */}

   398→          >
   399→            <span className="flex items-center gap-1"><Play size={14} /> MP4 HD</span>
   400→            <span className="flex items-center gap-1"><Music size={14} /> MP3 Audio</span>
   401→            <span className="flex items-center gap-1"><ImageIcon size={14} /> Cover Image</span>
   402→          </motion.div>
   403→        </div>
   404→      </section>
   405→
   406→      {/* ===== Video Result Section ===== */}
   407→      <AnimatePresence>
   408→        {videoInfo && (
   409→          <motion.section
   410→            initial={{ opacity: 0, y: 20 }}
   411→            animate={{ opacity: 1, y: 0 }}
   412→            exit={{ opacity: 0, y: 20 }}
   413→            className="max-w-4xl mx-auto px-4 sm:px-6 py-6 sm:py-8"
   414→          >
   415→            <div className="glass rounded-3xl p-4 sm:p-8">
   416→              {/* Tab selector */}
   417→              <div className="flex gap-2 mb-6">
   418→                {(['video', 'audio', 'cover'] as const).map((tab) => (
   419→                  <button
   420→                    key={tab}
   421→                    onClick={() => setActiveTab(tab)}
   422→                    className={`px-4 py-2 rounded-xl text-sm font-medium transition-colors ${
   423→                      activeTab === tab
   424→                        ? 'bg-[#FE2C55] text-white'
   425→                        : 'bg-white/10 text-gray-400 hover:bg-white/20'
   426→                    }`}
   427→                  >
   428→                    {tab === 'video' && <Play size={16} className="inline mr-1" />}
   429→                    {tab === 'audio' && <Music size={16} className="inline mr-1" />}
   430→                    {tab === 'cover' && <ImageIcon size={16} className="inline mr-1" />}
   431→                    {tab.charAt(0).toUpperCase() + tab.slice(1)}

   436→              <div className="flex flex-col lg:flex-row gap-6 sm:gap-8">
   437→                {/* Preview */}
   438→                <div className="lg:flex-1">

   440→                    <div className="relative rounded-2xl overflow-hidden bg-zinc-900">

   446→                      />
   447→                      <div className="absolute bottom-3 right-3 bg-black/70 px-2 py-1 rounded-lg text-xs flex items-center gap-1">
   448→                        <Clock size={12} /> {videoInfo.duration}
   449→                      </div>
   450→                    </div>
   451→                  )}
   452→                  {activeTab === 'audio' && (
   453→                    <div className="rounded-2xl bg-zinc-900 p-6 flex flex-col items-center justify-center min-h-[200px]">
   454→                      <Music size={48} className="text-[#25F4EE] mb-4" />
   455→                      <p className="text-gray-400 text-sm">Audio Preview</p>
   456→                      <p className="text-lg font-semibold mt-2">{videoInfo.title}</p>
   457→                    </div>
   458→                  )}
   459→                  {activeTab === 'cover' && (
   460→                    <div className="rounded-2xl overflow-hidden">
   461→                      <img
   462→                        src={videoInfo.cover || videoInfo.thumbnail}
   463→                        alt="Cover image"
   464→                        className="w-full object-cover"
   465→                      />
   466→                    </div>
   467→                  )}
   468→                </div>
   469→
   470→                {/* Info + download buttons */}
   471→                <div className="lg:flex-1 space-y-4 sm:space-y-6">
   472→                  <h3 className="text-xl sm:text-2xl font-semibold line-clamp-2">{videoInfo.title}</h3>
   473→
   474→                  <div className="flex items-center gap-3">
   475→                    {videoInfo.avatar && (
   476→                      <img
   477→                        src={videoInfo.avatar}
   478→                        alt={videoInfo.author}
   479→                        className="w-10 h-10 sm:w-12 sm:h-12 rounded-full object-cover bg-zinc-800"
   480→                        onError={(e) => { (e.target as HTMLImageElement).style.display = 'none'; }}
   481→                      />
   482→                    )}
   483→                    <div>
   484→                      <div className="font-medium">{videoInfo.author}</div>
   485→                      <div className="text-sm text-gray-500 flex items-center gap-3">
   486→                        {videoInfo.views && <span className="flex items-center gap-1"><User size={14} /> {videoInfo.views}</span>}
   487→                        {videoInfo.likes && <span className="flex items-center gap-1"><Heart size={14} /> {videoInfo.likes}</span>}
   488→                      </div>
   489→                    </div>
   490→                  </div>
   491→
   492→                  {/* Download actions */}
   493→                  <div className="space-y-3">
   494→                    {activeTab === 'video' && (
   495→                      <>
   496→                        <button
   497→                          onClick={() => handleDownload(videoInfo.noWatermarkUrl, getDownloadFilename('video', videoInfo))}
   498→                          className="w-full bg-[#FE2C55] hover:bg-[#FE2C55]/90 py-4 rounded-2xl font-semibold flex items-center justify-center gap-2 transition-colors"

   503→                        {videoInfo.withWatermarkUrl && !videoInfo.withWatermarkUrl.startsWith('#') && (

   505→                            onClick={() => handleDownload(videoInfo.withWatermarkUrl, `tiktok_${videoInfo.id}_with_watermark.mp4`)}
   506→                            className="w-full bg-white/10 hover:bg-white/20 py-3 rounded-2xl font-medium flex items-center justify-center gap-2 transition-colors"
   507→                          >
   508→                            <Download size={18} /> With Watermark
   509→                          </button>
   510→                        )}
   511→                      </>
   512→                    )}
   513→                    {activeTab === 'audio' && (
