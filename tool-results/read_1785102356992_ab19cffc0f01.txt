   127→// ===== Dynamic dimension parser =====
   128→function parseDimensions(dimStr: string): { w: number; h: number } {
   129→  if (dimStr === 'responsive' || dimStr === 'native') return { w: 300, h: 150 };
   130→  const parts = dimStr.split('x');
   131→  return { w: parseInt(parts[0]) || 300, h: parseInt(parts[1]) || 250 };
   132→}
