   108→const TikTokDownloader = () => {
   109→  const [url, setUrl] = useState('');
   110→  const [isLoading, setIsLoading] = useState(false);
   111→  const [videoInfo, setVideoInfo] = useState<VideoInfo | null>(null);
   112→  const [error, setError] = useState('');
   113→  const [history, setHistory] = useState<VideoInfo[]>([]);
   114→  const [activeTab, setActiveTab] = useState<'video' | 'audio' | 'cover'>('video');
   115→  const [showAdPopup, setShowAdPopup] = useState(false);
   116→  const [countdown, setCountdown] = useState(5);
   117→  const [pendingUrl, setPendingUrl] = useState('');
   118→  const [openFAQ, setOpenFAQ] = useState<number | null>(null);
   119→  const [autoProceedDone, setAutoProceedDone] = useState(false);
   120→  const [interstitialConfig, setInterstitialConfig] = useState({
   121→    enabled: true,
   122→    countdownDuration: 5,
   123→    autoDownload: true,
   124→    popupTitle: 'Support free downloads',
   125→    popupDescription: 'Your download will start automatically...',
   126→  });
   127→  const adTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   128→  const countdownRef = useRef<number>(5);
   129→  const autoProceedRef = useRef(false);
   130→  const inputRef = useRef<HTMLInputElement>(null);
   131→
   132→  // Fetch interstitial config on mount
   133→  useEffect(() => {
   134→    const fetchConfig = async () => {
   135→      try {
   136→        const res = await fetch('/api/config/ads');
   137→        const data = await res.json();
   138→        if (data.success && data.interstitial) {
   139→          setInterstitialConfig(data.interstitial);
   140→        }
   141→      } catch {
   142→        // Use default config on fetch failure
   143→      }
   144→    };
   145→    fetchConfig();
   146→  }, []);
   147→
   148→  // Load history from localStorage
   149→  useEffect(() => {
   150→    try {
   151→      const saved = localStorage.getItem('tiktokHistory');
   152→      if (saved) {
   153→        const parsed = JSON.parse(saved);
   154→        if (Array.isArray(parsed)) setHistory(parsed);
   155→      }
   156→    } catch (e) {
   157→      console.warn('Failed to load history:', e);
   158→    }
   159→  }, []);
   160→
   161→  // Save history to localStorage
   162→  useEffect(() => {
   163→    try {
   164→      localStorage.setItem('tiktokHistory', JSON.stringify(history));
   165→    } catch (e) {
   166→      console.warn('Failed to save history:', e);
   167→    }
   168→  }, [history]);
   169→

   171→    const regex = /^https?:\/\/(?:www\.|vm\.|vt\.|m\.)?tiktok\.com\/.+/i;
   172→    return regex.test(inputUrl.trim());

   176→    const duration = interstitialConfig.countdownDuration || 5;
   177→    setCountdown(duration);
   178→    setAutoProceedDone(false);
   179→    autoProceedRef.current = false;
   180→    setShowAdPopup(true);
   181→    setPendingUrl(videoUrl);
   182→    countdownRef.current = duration;
   183→
   184→    if (adTimerRef.current) clearInterval(adTimerRef.current);
   185→
   186→    adTimerRef.current = setInterval(() => {
   187→      countdownRef.current -= 1;
   188→      setCountdown(countdownRef.current);
   189→
   190→      if (countdownRef.current <= 0) {
   191→        if (adTimerRef.current) clearInterval(adTimerRef.current);
   192→        adTimerRef.current = null;
   193→      }
   194→    }, 1000);
   195→  }, [interstitialConfig.countdownDuration]);
   196→

   198→    setShowAdPopup(false);
   199→
   200→    const videoUrl = pendingUrl;
   201→    if (!videoUrl) return;
   202→
   203→    setIsLoading(true);
   204→    setError('');
   205→    setVideoInfo(null);
   206→
   207→    try {
   208→      const response = await fetch('/api/download', {
   209→        method: 'POST',
   210→        headers: { 'Content-Type': 'application/json' },
   211→        body: JSON.stringify({ url: videoUrl }),
   212→      });
   213→
   214→      const result = await response.json();
   215→
   216→      if (!result.success) {
   217→        throw new Error(result.error || 'Failed to fetch video info');
   218→      }
   219→
   220→      const data: VideoInfo = result.data;
   221→      setVideoInfo(data);
   222→      setHistory(prev => [data, ...prev.slice(0, 4)]);
   223→
   224→      toast.success('Video ready!', {
   225→        description: `Fetched via ${result.provider} in ${result.duration}ms`,
   226→      });
   227→    } catch (err: unknown) {
   228→      const errorMessage = err instanceof Error ? err.message : 'Unable to process this video. Try again.';
   229→      console.error('Download error:', err);
   230→      setError(errorMessage);
   231→      toast.error('Fetch failed', { description: errorMessage });
   232→    } finally {
   233→      setIsLoading(false);
   234→      setPendingUrl('');
   235→    }
   236→  }, [pendingUrl]);
   237→
   238→  // Auto-proceed when countdown reaches 0
   239→  useEffect(() => {
   240→    if (countdown === 0 && showAdPopup && interstitialConfig.autoDownload && !autoProceedRef.current) {
   241→      autoProceedRef.current = true;
   242→      setAutoProceedDone(true);
   243→      const timeout = setTimeout(() => {
   244→        proceedAfterAd();
   245→      }, 800);
   246→      return () => clearTimeout(timeout);
   247→    }
   248→  }, [countdown, showAdPopup, interstitialConfig.autoDownload, proceedAfterAd]);
   249→
   250→  const handleSubmit = useCallback((e: React.FormEvent) => {
   251→    e.preventDefault();
   252→    const trimmed = url.trim();

   254→      toast.error('Enter a TikTok URL');

   256→    }
   257→    if (!isValidTikTokUrl(trimmed)) {

   259→      toast.error('Please use a valid TikTok link');
   260→      return;
   261→    }
   262→    setError('');
   263→    startAdTimer(trimmed);
   264→  }, [url, startAdTimer]);
   265→
   266→  const handleDownload = useCallback((downloadUrl: string, filename: string) => {
   267→    if (!downloadUrl || downloadUrl.startsWith('#')) {
   268→      toast.error('Download URL not available for this video.');
   269→      return;
   270→    }
   271→
   272→    const a = document.createElement('a');
   273→    a.href = downloadUrl;
   274→    a.download = filename;
   275→    a.target = '_blank';
   276→    a.rel = 'noopener noreferrer';
   277→    document.body.appendChild(a);
   278→    a.click();
   279→    document.body.removeChild(a);
   280→
   281→    toast.success(`Downloading ${filename}`, {
   282→      description: 'Your file download has started.',
   283→    });
   284→  }, []);
   285→
   286→  const getDownloadFilename = useCallback((type: 'video' | 'audio' | 'cover', info: VideoInfo): string => {
   287→    const sanitizeName = (name: string) => name.replace(/[^a-zA-Z0-9_-]/g, '_').slice(0, 50);
   288→    const baseName = sanitizeName(info.title || info.id);
   289→    switch (type) {
   290→      case 'video': return `tiktok_${baseName}_no_watermark.mp4`;
   291→      case 'audio': return `tiktok_${baseName}_audio.mp3`;
   292→      case 'cover': return `tiktok_${baseName}_cover.jpg`;
   293→    }
   294→  }, []);
   295→
   296→  const copyToClipboard = useCallback((text: string, label: string) => {
   297→    navigator.clipboard.writeText(text).then(() => {
   298→      toast.success(`${label} copied!`);
   299→    }).catch(() => {
   300→      toast.error('Failed to copy');
   301→    });
   302→  }, []);
   303→
   304→  const clearHistory = useCallback(() => {
   305→    setHistory([]);
   306→    localStorage.removeItem('tiktokHistory');
   307→    toast.success('History cleared');
   308→  }, []);
   309→
   310→  // Cleanup timer on unmount
   311→  useEffect(() => {
   312→    return () => {
   313→      if (adTimerRef.current) clearInterval(adTimerRef.current);
   314→    };
   315→  }, []);
   316→
   317→  // Handle paste from clipboard
   318→  const handlePaste = useCallback(async () => {
   319→    try {
   320→      const text = await navigator.clipboard.readText();
   321→      setUrl(text);
   322→      setError('');
   323→      toast.success('Link pasted from clipboard');
   324→      // Hide mobile keyboard
   325→      inputRef.current?.blur();
   326→    } catch {
   327→      toast.error('Cannot read clipboard — please paste manually');
   328→    }
   329→  }, []);
   330→
   331→  // Handle clear input
   332→  const handleClearInput = useCallback(() => {
   333→    setUrl('');
   334→    setError('');
   335→    inputRef.current?.focus();
   336→  }, []);
   337→
   338→  // Countdown ring calculation
   339→  const circumference = 2 * Math.PI * 36;
   340→  const progress = interstitialConfig.countdownDuration > 0
   341→    ? (interstitialConfig.countdownDuration - countdown) / interstitialConfig.countdownDuration
   342→    : 1;
   343→  const strokeDashoffset = circumference * (1 - progress);
   344→
   345→  return (
   346→    <div className="min-h-screen bg-[#000000] text-white flex flex-col">
   347→      <Toaster position="top-center" richColors closeButton />
   348→
   349→      {/* ===== Navbar ===== */}
   350→      <nav className="sticky top-0 z-50 glass border-b border-white/10 bg-black/80">
   351→        <div className="max-w-6xl mx-auto px-4 sm:px-6 py-3 flex justify-between items-center">
   352→          <div className="flex items-center gap-2.5">
   353→            <div className="w-7 h-7 bg-[#FE2C55] rounded-lg flex items-center justify-center text-base font-bold">♪</div>
   354→            <span className="font-bold text-xl tracking-tighter">TikDL</span>
   355→          </div>
   356→          <div className="hidden md:flex gap-6 text-sm font-medium text-gray-400">

   358→            <a href="#faq" className="hover:text-[#FE2C55] transition-colors duration-150">FAQ</a>
   359→            <a href="#history" className="hover:text-[#FE2C55] transition-colors duration-150">History</a>
   360→          </div>

   364→      {/* ===== Hero Section ===== */}

   367→          {/* Badge */}
   368→          <motion.div
   369→            initial={{ opacity: 0, y: 10 }}
   370→            animate={{ opacity: 1, y: 0 }}
   371→            transition={{ duration: 0.4 }}
   372→            className="inline-flex items-center gap-2 px-5 py-2 rounded-full border border-[rgba(254,44,85,0.3)] text-[#FE2C55] text-sm font-semibold mb-8 mt-6"
   373→          >
   374→            <Sparkles size={14} />
   375→            Free and Unlimited

   378→          {/* Hero heading */}
   379→          <motion.h1
   380→            initial={{ opacity: 0, y: 20 }}
   381→            animate={{ opacity: 1, y: 0 }}
   382→            transition={{ duration: 0.5, delay: 0.1 }}
   383→            className="text-[clamp(36px,5vw,48px)] font-extrabold tracking-tight leading-[1.1] mb-6"
   384→          >
   385→            <span className="text-white">TikTok Video</span>
   386→            <br />
   387→            <span className="text-[#FE2C55]">Without Watermark</span>
   388→          </motion.h1>
   389→
   390→          {/* Description */}
   391→          <motion.p
   392→            initial={{ opacity: 0, y: 10 }}
   393→            animate={{ opacity: 1, y: 0 }}

   395→            className="text-base text-[#9CA3AF] leading-relaxed max-w-[400px] mx-auto mb-10"

   415→                disabled={isLoading || showAdPopup}
   416→              />
   417→              {/* Paste/Clear toggle button */}
   418→              {url ? (

   423→                  title="Clear input"

   433→                  title="Paste from clipboard"

   482→              <Play size={16} className="text-[#FE2C55]" /> MP4 HD

   490→          </motion.div>
   491→        </div>

   494→      {/* ===== Video Result Section ===== */}

   500→            exit={{ opacity: 0, y: 20 }}

   524→              <div className="flex flex-col lg:flex-row gap-5">
